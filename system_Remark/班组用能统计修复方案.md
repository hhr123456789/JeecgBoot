# 班组用能统计功能修复方案

## 一、问题总结

### 根本原因
1. **数据库**: `tb_team_dimension_relation.module_ids` 全部为空，导致班组无法关联仪表
2. **后端**: 4个核心接口（getStatistics/getTrendData/getRankingData/getTableData）返回硬编码模拟数据，未实现真实查询
3. **前端**: DimensionTree 字段映射问题，用户交互后不自动刷新

### 影响
即使后端代码正确实现，由于数据库关联缺失，也无法查询到任何能耗数据。

## 二、修复策略

### 用户决策
- **数据关联**: 按部门自动关联（根据 dimension_code 关联该部门下的所有仪表）
- **排班处理**: 不依赖排班时间，按整天统计
- **修复范围**: 全部修复（数据 + 后端 + 前端）

## 三、详细修复方案

### 阶段 1: 数据库修复

#### 1.1 修复 dimension_code
**问题**: 当前 `A01B03` 和 `A01B03C01` 在 sys_depart 中不存在

**解决方案**:
- 查询 sys_depart 表，找到有效的部门编码
- 将 tb_team_dimension_relation 中的 dimension_code 更新为有效值
- 或者在 sys_depart 中创建对应的部门记录

**SQL 示例**:
```sql
-- 方案1: 更新为已存在的部门编码（需要先确认映射关系）
UPDATE tb_team_dimension_relation
SET dimension_code = 'A01'
WHERE dimension_code = 'A01B03';

-- 方案2: 在 sys_depart 中创建对应部门
INSERT INTO sys_depart (id, org_code, depart_name, org_category, status, del_flag)
VALUES
  (UUID(), 'A01B03', '部门B03', '2', '1', '0'),
  (UUID(), 'A01B03C01', '部门C01', '2', '1', '0');
```

#### 1.2 填充 module_ids
**目标**: 为每个班组关联对应部门下的仪表

**实现逻辑**:
1. 查询每个 dimension_code 对应的部门
2. 查询该部门下所有启用的仪表（isaction='Y'）
3. 将仪表 ID 用逗号连接，更新到 module_ids 字段

**SQL**:
```sql
-- 为每个班组维度关联分配仪表
UPDATE tb_team_dimension_relation tdr
SET module_ids = (
    SELECT GROUP_CONCAT(m.module_id SEPARATOR ',')
    FROM tb_module m
    WHERE m.sys_org_code = tdr.dimension_code
      AND m.isaction = 'Y'
      AND m.energy_type = tdr.energy_type
)
WHERE (module_ids IS NULL OR module_ids = '')
  AND status = 1;
```

**验证**:
```sql
-- 检查更新结果
SELECT
    team_code,
    dimension_code,
    energy_type,
    module_ids,
    LENGTH(module_ids) - LENGTH(REPLACE(module_ids, ',', '')) + 1 as module_count
FROM tb_team_dimension_relation
WHERE status = 1;
```

```java
// 辅助方法：获取班组关联的仪表ID列表
private List<String> getModuleIdsByTeamCode(String teamCode, String dimensionCode) {
    LambdaQueryWrapper<TeamDimensionRelation> wrapper = new LambdaQueryWrapper<>();

    if (!"all".equals(teamCode)) {
        wrapper.eq(TeamDimensionRelation::getTeamCode, teamCode);
    }

    if (StringUtils.isNotBlank(dimensionCode)) {
        wrapper.eq(TeamDimensionRelation::getDimensionCode, dimensionCode);
    }

    wrapper.eq(TeamDimensionRelation::getStatus, 1);

    List<TeamDimensionRelation> relations = teamDimensionRelationMapper.selectList(wrapper);

    // 解析 module_ids（逗号分隔）
    return relations.stream()
        .map(TeamDimensionRelation::getModuleIds)
        .filter(StringUtils::isNotBlank)
        .flatMap(ids -> Arrays.stream(ids.split(",")))
        .map(String::trim)
        .filter(StringUtils::isNotBlank)
        .distinct()
        .collect(Collectors.toList());
}
```

##### 2.1.2 getTrendData() - 趋势数据
```java
@Override
public TeamEnergyTrendVO getTrendData(TeamEnergyQueryRequest request) {
    TeamEnergyTrendVO vo = new TeamEnergyTrendVO();

    // 1. 生成 X 轴数据（日期列表）
    List<String> xAxisData = generateXAxisData(request.getTimeUnit(),
                                                request.getStartDate(),
                                                request.getEndDate());
    vo.setXAxisData(xAxisData);

    // 2. 查询班组列表
    List<TeamInfo> teams = getTeamList(request.getTeamCode(), request.getDimensionCode());

    // 3. 为每个班组查询趋势数据
    List<TeamEnergyTrendVO.SeriesData> seriesList = new ArrayList<>();

    for (TeamInfo team : teams) {
        List<String> moduleIds = getModuleIdsByTeamCode(team.getTeamCode(),
                                                        request.getDimensionCode());
        if (moduleIds.isEmpty()) {
            continue;
        }

        // 查询该班组在时间范围内的数据
        List<BigDecimal> dataValues = new ArrayList<>();
        for (String date : xAxisData) {
            BigDecimal value = queryEnergyByDate(moduleIds, date, request.getTimeUnit());
            dataValues.add(value);
        }

        TeamEnergyTrendVO.SeriesData series = new TeamEnergyTrendVO.SeriesData();
        series.setName(team.getTeamName());
        series.setData(dataValues.stream()
            .map(v -> v.setScale(2, RoundingMode.HALF_UP).toString())
            .collect(Collectors.toList()));
        series.setColor(getColorByIndex(seriesList.size()));

        seriesList.add(series);
    }

    vo.setSeriesData(seriesList);
    return vo;
}
```

```java
                                                   request.getEndDate(),
                                                   request.getTimeUnit());

        TeamEnergyRankingVO vo = new TeamEnergyRankingVO();
        vo.setName(team.getTeamName());
        vo.setValue(totalEnergy.setScale(2, RoundingMode.HALF_UP).toString());
        vo.setUnit("kWh");

        rankings.add(vo);

    rankings.sort((a, b) -> new BigDecimal(b.getValue())
        .compareTo(new BigDecimal(a.getValue())));

    return rankings;
}
```

```java
@Override
public List<TeamEnergyTableVO> getTableData(TeamEnergyQueryRequest request) {
    List<TeamEnergyTableVO> tableData = new ArrayList<>();

    // 1. 查询班组列表
    List<TeamInfo> teams = getTeamList(request.getTeamCode(), request.getDimensionCode());

    // 2. 为每个班组查询明细数据
    for (TeamInfo team : teams) {
        List<String> moduleIds = getModuleIdsByTeamCode(team.getTeamCode(),
                                                        request.getDimensionCode());
        if (moduleIds.isEmpty()) {
            continue;
        }

        // 查询能耗数据
        LambdaQueryWrapper<TbEpEquEnergyDaycount> wrapper = new LambdaQueryWrapper<>();
        wrapper.in(TbEpEquEnergyDaycount::getModuleId, moduleIds);
        wrapper.between(TbEpEquEnergyDaycount::getDt,
                       request.getStartDate(),
                       request.getEndDate());

        List<TbEpEquEnergyDaycount> dataList = daycountMapper.selectList(wrapper);

        // 按日期分组汇总
        Map<Date, List<TbEpEquEnergyDaycount>> groupedData = dataList.stream()
            .collect(Collectors.groupingBy(TbEpEquEnergyDaycount::getDt));

        for (Map.Entry<Date, List<TbEpEquEnergyDaycount>> entry : groupedData.entrySet()) {
            TeamEnergyTableVO vo = new TeamEnergyTableVO();
            vo.setTeamName(team.getTeamName());
            vo.setShiftType(team.getShiftType());
            vo.setStatTime(new SimpleDateFormat("yyyy-MM-dd").format(entry.getKey()));

            // 汇总计算
            List<TbEpEquEnergyDaycount> dayData = entry.getValue();
            vo.setConsumption(sumField(dayData, TbEpEquEnergyDaycount::getEnergyCount));
            vo.setCost(sumField(dayData, TbEpEquEnergyDaycount::getTotalCount));
            vo.setCarbon(sumField(dayData, TbEpEquEnergyDaycount::getTpfCount));
            vo.setCoal(sumField(dayData, TbEpEquEnergyDaycount::getZbmCount));
            vo.setPeak(sumField(dayData, TbEpEquEnergyDaycount::getPeakCount));
            vo.setFlat(sumField(dayData, TbEpEquEnergyDaycount::getLevelCount));
            vo.setValley(sumField(dayData, TbEpEquEnergyDaycount::getValleyCount));

            tableData.add(vo);
        }
    }

    return tableData;
}
```

```java
// 根据时间维度获取表名
private String getTableNameByTimeUnit(String timeUnit) {
    switch (timeUnit) {
        case "day": return "tb_ep_equ_energy_daycount";
        case "month": return "tb_ep_equ_energy_monthcount";
        case "year": return "tb_ep_equ_energy_yearcount";
        default: return "tb_ep_equ_energy_daycount";
    }
}

// 生成 X 轴数据
private List<String> generateXAxisData(String timeUnit, Date startDate, Date endDate) {
    List<String> xAxisData = new ArrayList<>();
    SimpleDateFormat sdf;

    switch (timeUnit) {
        case "day":
            sdf = new SimpleDateFormat("MM-dd");
            break;
        case "month":
            sdf = new SimpleDateFormat("yyyy-MM");
            break;
        case "year":
            sdf = new SimpleDateFormat("yyyy");
            break;
        default:
            sdf = new SimpleDateFormat("MM-dd");
    }

    Calendar calendar = Calendar.getInstance();
    calendar.setTime(startDate);

    while (!calendar.getTime().after(endDate)) {
        xAxisData.add(sdf.format(calendar.getTime()));

        switch (timeUnit) {
            case "day":
                calendar.add(Calendar.DAY_OF_MONTH, 1);
                break;
            case "month":
                calendar.add(Calendar.MONTH, 1);
                break;
            case "year":
                calendar.add(Calendar.YEAR, 1);
                break;
        }
    }

    return xAxisData;
}
```
